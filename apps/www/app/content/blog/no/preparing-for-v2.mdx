Da `@digdir/designsystemet-web` ble lansert i februar, tok designsystemet.no et viktig steg mot prosjektets visjon:
å tilby små, robuste byggeklosser basert på webstandarder, som er enkle å bruke og bygge videre på for alle.
V2 fortsetter i samme retning. Vi utvider fargesystemet, og fjerner gamle løsninger til fordel for webstandarder.

## Nye fargetrinn

Flere har meldt fra om utfordringer med at komponenter i dag "låner" farger på tvers av fargeskalaer.
For eksempel bruker input-komponenten farger fra `danger` når feltet er ugyldig, og fra `neutral` når feltet er i lesemodus.

Derfor får hver fargeskala noen nye fargetrinn: `surface-muted`, `border-muted`, `text-muted`, `surface-invalid` og `text-invalid`.
Da kan komponentene hente alle fargene sine fra én og samme skala, og det blir lettere å lage fargeskalaer uten uforutsette konsekvenser ved bruk.

## Slik forbereder du koden

Vi vet at breaking changes tar tid å håndtere, og det er viktig for oss at overgangen blir smidig.
Slik kan du gjøre deg klar allerede nå:

1. **Bruk siste 1.x-versjon** av `@digdir/designsystemet*`-pakkene.
2. **Se etter advarsler:** TypeScript markerer utgåtte props i editor, React- og web-pakken skriver advarsler i konsoll, og CLI varsler i terminal.
3. **Bytt ut utgått kode:** Koden i listene under er allerede markert som `@deprecated` i v1, og kan byttes ut uten å vente på v2.

<Details.Summary>Kode for `useDebounceCallback`</Details.Summary>

```ts
import { useEffect, useRef } from 'react';

type DebounceFunction<T> = (...args: T[]) => void;

export function useDebounceCallback<T>(
  callback: DebounceFunction<T>,
  delay = 50,
) {
  const timeoutRef = useRef<number | null>(null);

  useEffect(() => {
    // Cleanup the previous timeout on re-render
    return () => {
      if (timeoutRef.current) {
        clearTimeout(timeoutRef.current);
      }
    };
  }, []);

  const debouncedCallback = (...args: T[]) => {
    if (timeoutRef.current) {
      clearTimeout(timeoutRef.current);
    }

    timeoutRef.current = window.setTimeout(() => {
      callback(...args);
    }, delay);
  };

  return debouncedCallback;
}
```

<Details.Summary>Kode for `useSynchronizedAnimation`</Details.Summary>

```ts
// Inspired by Sam Selikoff
// https://github.com/samselikoff/2022-02-24-use-synchronized-animation/blob/main/src/App.js
import { useEffect, useLayoutEffect, useRef } from 'react';

/**
 * Synchronizes the css animation of multiple elements with the same `animationName`.
 *
 * @example
 * const ref = useSynchronizedAnimation<HTMLDivElement>('spin');
 *
 * <div ref={ref} />
 */
export function useSynchronizedAnimation<T>(animationName: string) {
  const ref = useRef<T>(null);
  const useHook = typeof window === 'undefined' ? useEffect : useLayoutEffect;

  useHook(() => {
    const animations = document
      .getAnimations()
      .filter(
        (animation) =>
          'animationName' in animation &&
          animation.animationName === animationName,
      );

    const firstOfType = animations.find(
      (animation) =>
        'animationName' in animation &&
        animation.animationName === animationName,
    );

    const myAnimation = animations.find(
      (animation) =>
        (animation.effect as KeyframeEffect)?.target === ref.current,
    );

    if (myAnimation && myAnimation === firstOfType) {
      myAnimation.currentTime = 0;
    }

    if (
      myAnimation &&
      firstOfType?.currentTime &&
      myAnimation !== firstOfType
    ) {
      myAnimation.currentTime = firstOfType.currentTime;
    }

    return () => {
      if (myAnimation && firstOfType?.currentTime) {
        myAnimation.currentTime = firstOfType.currentTime;
      }
    };
  }, [animationName]);

  return ref;
}
```

<Details.Summary>Kode for `useMediaQuery`</Details.Summary>

```ts
import { useEffect, useLayoutEffect, useState } from 'react';

type UseMediaQueryOptions = {
  defaultValue?: boolean;
  initializeWithValue?: boolean;
};

export function useMediaQuery(
  query: string,
  {
    defaultValue = false,
    initializeWithValue = true,
  }: UseMediaQueryOptions = {},
): boolean {
  const useHook = typeof window === 'undefined' ? useEffect : useLayoutEffect;
  const onChange = () => setMatches(getMatches(query));
  const getMatches = (query: string): boolean => {
    if (typeof window === 'undefined') return defaultValue;
    return window.matchMedia(query).matches;
  };
  const [matches, setMatches] = useState<boolean>(() => {
    return initializeWithValue ? defaultValue : getMatches(query);
  });

  useHook(() => {
    const matchMedia = window.matchMedia(query);
    onChange(); // Triggered at the first client-side load and if query changes

    // Use deprecated `addListener` and `removeListener` to support Safari < 14
    if (matchMedia.addListener) matchMedia.addListener(onChange);
    else matchMedia.addEventListener('change', onChange);

    return () => {
      if (matchMedia.removeListener) matchMedia.removeListener(onChange);
      else matchMedia.removeEventListener('change', onChange);
    };
  }, [query]);

  return matches;
}
```

```ts
type UnknownRecord = Record<string | number | symbol, unknown>;

/**
 * Returns a partial copy of an object omitting the keys specified.
 * @example omit(['a', 'd'], {a: 1, b: 2, c: 3, d: 4}); //=> {b: 2, c: 3}
 */
export const omit = (names: string[], obj: UnknownRecord): object => {
  const result: UnknownRecord = {};
  const index: UnknownRecord = {};
  let idx = 0;
  const len = names.length;

  while (idx < len) {
    index[names[idx]] = 1;
    idx += 1;
  }

  for (const prop in obj)
    if (!Object.hasOwn(index, prop))
      result[prop] = obj[prop];

  return result;
};
```
